`timescale 1ns / 1ps
//////////////////////////////////////////////////////////////////////////////////
// Top Level Module: Real-Time Stream Cipher Engine
// Target Board: PYNQ-Z2 (Zynq-7000 SoC)
// Description: XOR-based symmetric encryption/decryption engine.
//////////////////////////////////////////////////////////////////////////////////

module top_crypto (
    input wire clk,           // 125 MHz System Clock
    input wire [1:0] sw,      // Plaintext inputs from SW0 and SW1
    input wire [1:0] btn,     // btn[0]: Reset, btn[1]: LFSR Step/Enable
    output wire [3:0] led     // led[1:0]: Ciphertext, led[3:2]: Decrypted Plaintext
);

    wire [7:0] key_stream;
    
    // Instantiate 8-bit LFSR Key Generator
    lfsr_8bit key_gen (
        .clk(clk),
        .reset(btn[0]),
        .enable(btn[1]),
        .lfsr_out(key_stream)
    );

    // --- Cryptographic Engine ---
    // Encryption: Ciphertext = Plaintext XOR Key Stream
    wire [1:0] ciphertext;
    assign ciphertext = sw ^ key_stream[1:0];

    // Decryption: Recovered Plaintext = Ciphertext XOR Key Stream
    wire [1:0] decrypted_plaintext;
    assign decrypted_plaintext = ciphertext ^ key_stream[1:0];

    // Hardware Output Mapping
    assign led[1:0] = ciphertext;           // Displays active data scrambling
    assign led[3:2] = decrypted_plaintext;  // Displays recovered input data

endmodule
